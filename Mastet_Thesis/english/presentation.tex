\begin{frame}
	\frametitle{Pauli Operators}

\begin{equation}
	\hat{I} = \begin{pmatrix} 
		1 & 0 \\ 
		0 & 1 
	\end{pmatrix}
	\label{eq:matrix_I}
\end{equation}

\begin{equation}
	\hat{X} = \begin{pmatrix} 
		0 & 1 \\ 
		1 & 0 
	\end{pmatrix}
	\label{eq:matrix_X}
\end{equation}

\begin{equation}
	\hat{Y} = \begin{pmatrix} 
		0 & -i \\ 
		i & 0 
	\end{pmatrix}
	\label{eq:matrix_Y}
\end{equation}

\begin{equation}
	\hat{Z} = \begin{pmatrix} 
		1 & 0 \\ 
		0 & -1 
	\end{pmatrix}
	\label{eq:matrix_Z}
\end{equation}
